// Grade id S2408 // list all arrays // setup the shipping provider // set up for the contract // setup discount and savings // declare list of varibles // const of vairbles // valdiate comboboxs // calculations // output to labels using System; using System.Collections.Generic; using System.ComponentModel; using System.Data; using System.Drawing; using System.Linq; using System.Text; using System.Threading.Tasks; using System.Windows.Forms; namespace Program3 { public partial class Contract : Form { // lists of all arrays string[] shippingCompanies = { "USPS", "DHL", "FedEx", "UPS" }; double[] DiscountRates = { 0.25, 0.20, 0.17, 0.15 }; string[] businessNames = { "John's Books", "Office Supplies", "J.B. Car Parts", "Gevalia Coffee", "Ceylon Tea", "My Footwear" }; double[] initialPrices = { 55000, 87880, 65000, 90000, 75500, 40550 }; double[] contractLengthLowEnd = { 0, 2, 5, 8 }; double[] savings = { 0, 5000, 10000, 15000 }; //combo boxes output forms public Contract() { InitializeComponent(); InitializeComboBoxes(); } private void InitializeComboBoxes() { foreach (var company in shippingCompanies) { shipComboBox.Items.Add(company); } foreach (var business in businessNames) { bizComboBox.Items.Add(business); } } private void calculateBtn_Click(object sender, EventArgs e) { // valdiate comboboxs if (shipComboBox.SelectedIndex < 0 || bizComboBox.SelectedIndex < 0) { MessageBox.Show("Please select a provider and business."); return; } if (!double.TryParse(contractLengthTb.Text, out double contractLength) || contractLength < 0) { MessageBox.Show("Please enter a valid contract length in years."); return; } int shipIdx = shipComboBox.SelectedIndex; int bizIdx = bizComboBox.SelectedIndex; double initialPrice = initialPrices[bizIdx]; double companyDiscount = DiscountRates[shipIdx]; // calculate length discount based on contract length ranges double lengthDiscount = 0; double lengthSavings = 0; for (int i = contractLengthLowEnd.Length - 1; i >= 0; i--) { if (contractLength >= contractLengthLowEnd[i]) { lengthSavings = savings[i]; lengthDiscount = i * 0.02; break; } } double discountedPrice = initialPrice * (1 - companyDiscount); double finalPrice = discountedPrice - lengthSavings; double totalSavings = initialPrice - finalPrice; // output to labels providerLbl.Text = shippingCompanies[shipIdx]; initialPriceLbl.Text = initialPrice.ToString("C"); compDiscountLbl.Text = (companyDiscount * 100).ToString("F0") + "%"; lengthDiscLbl.Text = lengthSavings.ToString("C"); finalPriceLbl.Text = finalPrice.ToString("C"); totalSavingsLbl.Text = totalSavings.ToString("C"); } } }
