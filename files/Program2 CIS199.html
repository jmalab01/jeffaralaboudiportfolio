using System; using System.Collections.Generic; using System.ComponentModel; using System.Data; using System.Drawing; using System.Linq; using System.Text; using System.Threading.Tasks; using System.Windows.Forms; namespace program2 { public partial class Form1 : Form { public Form1() { InitializeComponent(); } private void button1_Click(object sender, EventArgs e) { const int MinimumAge = 18; const int MaximumAge = 120; // constant variables for exercise point values const int RunningPoints = 10; const int WeightTrainingPoints = 20; const int CyclingPoints = 5; const int YogaPoints = 2; const int OtherPoints = 5; typepick.Items.AddRange(new object[] { "Running", "Weight Training", "Cycling", "Yoga", "Other" }); string name = nametb.Text.Trim(); int age; // age using constant variables if (!int.TryParse(agetb.Text, out age) || age < MinimumAge || age > MaximumAge) { MessageBox.Show("Please enter a valid age between 18 and 120.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error); return; } if (typepick.SelectedIndex < 0) { MessageBox.Show("Please select an exercise type.", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error); return; } // calculate points based on exercise type and duration int duration; int.TryParse(durationtb.Text, out duration); int pointsPerMin = typepick.SelectedItem.ToString() switch { "Running" => RunningPoints, "Weight Training" => WeightTrainingPoints, "Cycling" => CyclingPoints, "Yoga" => YogaPoints, _ => OtherPoints }; int points = pointsPerMin * duration; bool hydrated = hydratedYes.Checked; bool smokes = smokesYes.Checked; if (hydrated) points += 5; if (smokes) points -= 10; string advice = smokes ? "Quitting smoking will greatly improve your fitness results!" : hydrated ? "Great job staying hydrated! Keep it up." : "Remember to drink water during exercise."; pointstb.Text = points.ToString(); advicetb.Text = advice; } } }
