Entity-Relationship Overview 6 Tables

The STAYWELL database models a property management company with two Seattle-area offices. It tracks property owners, rental properties, office locations, residents, service categories, and maintenance requests.

  • OFFICE → PROPERTY  (one office manages many properties via OFFICE_NUM)
  • OWNER → PROPERTY  (one owner has many properties via OWNER_NUM)
  • PROPERTY → RESIDENTS  (one property has many residents via PROPERTY_ID)
  • PROPERTY → SERVICE_REQUEST  (one property has many service requests via PROPERTY_ID)
  • SERVICE_CATEGORY → SERVICE_REQUEST  (one category applies to many requests via CATEGORY_NUM)

-- ══ STAYWELL Database Schema (SQL Server) ══ BEGIN TRANSACTION; -- OWNER CREATE TABLE OWNER ( OWNER_NUM CHAR(5) PRIMARY KEY NOT NULL, LAST_NAME VARCHAR(20) NOT NULL, FIRST_NAME VARCHAR(20) NOT NULL, ADDRESS VARCHAR(100) NOT NULL, CITY VARCHAR(20) NOT NULL, STATE CHAR(2) NOT NULL, ZIP_CODE CHAR(5) NOT NULL ); -- OFFICE CREATE TABLE OFFICE ( OFFICE_NUM TINYINT PRIMARY KEY NOT NULL, OFFICE_NAME VARCHAR(50) NOT NULL, ADDRESS VARCHAR(100) NOT NULL, AREA VARCHAR(50) NOT NULL, CITY VARCHAR(20) NOT NULL, STATE CHAR(2) NOT NULL, ZIP_CODE CHAR(5) NOT NULL ); -- PROPERTY CREATE TABLE PROPERTY ( PROPERTY_ID SMALLINT PRIMARY KEY NOT NULL, OFFICE_NUM TINYINT NOT NULL, -- FK → OFFICE ADDRESS VARCHAR(100) NOT NULL, SQR_FT SMALLINT NOT NULL, BDRMS TINYINT NOT NULL, FLOORS TINYINT NOT NULL, MONTHLY_RENT SMALLINT, OWNER_NUM CHAR(5) NOT NULL -- FK → OWNER ); -- SERVICE_CATEGORY CREATE TABLE SERVICE_CATEGORY ( CATEGORY_NUM TINYINT PRIMARY KEY NOT NULL, CATEGORY_DESCRIPTION VARCHAR(100) NOT NULL ); -- SERVICE_REQUEST CREATE TABLE SERVICE_REQUEST ( SERVICE_ID SMALLINT PRIMARY KEY NOT NULL, PROPERTY_ID SMALLINT NOT NULL, -- FK → PROPERTY CATEGORY_NUMBER TINYINT NOT NULL, -- FK → SERVICE_CATEGORY OFFICE_ID TINYINT NOT NULL, -- FK → OFFICE DESCRIPTION VARCHAR(200) NOT NULL, STATUS VARCHAR(200) NOT NULL, EST_HOURS TINYINT NOT NULL, SPENT_HOURS TINYINT NOT NULL, NEXT_SERVICE_DATE DATE ); -- RESIDENTS CREATE TABLE RESIDENTS ( RESIDENT_ID SMALLINT PRIMARY KEY NOT NULL, FIRST_NAME VARCHAR(20) NOT NULL, SURNAME VARCHAR(20) NOT NULL, PROPERTY_ID SMALLINT NOT NULL -- FK → PROPERTY ); COMMIT TRANSACTION;
