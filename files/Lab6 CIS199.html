//Grade id s2408 // display the constant value for max row // the first pattern A forming a triangle shape, and prints it to the console // second pattern B forming a triangle shape, and prints it to the console // third pattern c forming a triangle shape, and prints it to the console // fourth pattern d forming a triangle shape, and prints it to the console using System; using System.Collections.Generic; using System.Linq; using System.Text; using System.Threading.Tasks; namespace Lab6 { internal class Program { static void Main(string[] args) { const int MAX_ROWS = 10; Console.WriteLine("Pattern A\n"); for (int row = 1; row <= MAX_ROWS; row++) { for (int dollarsign = 1; dollarsign <= row; dollarsign++) { Console.Write("$"); } Console.WriteLine(); } Console.WriteLine("\nPattern B\n"); for (int row = MAX_ROWS; row >= 1; row--) { for (int dollarsign = 1; dollarsign <= row; dollarsign++) { Console.Write("$"); } Console.WriteLine(); } Console.WriteLine("\nPattern C\n"); for (int row = MAX_ROWS; row >= 1; row--) { for (int spaces = 0; spaces < MAX_ROWS - row; spaces++) { Console.Write(" "); } for (int dollarsign = 1; dollarsign <= row; dollarsign++) { Console.Write("$"); } Console.WriteLine(); } Console.WriteLine("\nPattern D\n"); for (int row = 1; row <= MAX_ROWS; row++) { for (int spaces = 0; spaces < MAX_ROWS - row; spaces++) { Console.Write(" "); } for (int dollarsign = 1; dollarsign <= row; dollarsign++) { Console.Write("$"); } Console.WriteLine(); } } } }
