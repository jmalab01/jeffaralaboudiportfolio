Please enter a positive integer."); } } } static void ShowSquareOfStars(int numStars) { // Display the size of the square Console.WriteLine($"\n{numStars}x{numStars} Square of Asterisks\n"); // Loop through each row of the square for (int row = 0; row < numStars; row++) { // Loop through each column of the square for (int col = 0; col < numStars; col++) { Console.Write("* "); } Console.WriteLine(); } } } }
